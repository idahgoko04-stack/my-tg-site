    <script>
        // Инициализация Telegram Web App
        const tg = window.Telegram.WebApp;
        tg.expand(); // Развернуть на весь экран в ТГ

        // Извлечение имени пользователя из ТГ (если доступно)
        // [Интерпретировано]: Объект initDataUnsafe может быть пустым в обычном браузере, проверяем наличие данных.
        if (tg.initDataUnsafe && tg.initDataUnsafe.user) {
            const user = tg.initDataUnsafe.user;
            document.getElementById('greeting').innerText = `Привет, ${user.first_name}!`;
            if (user.first_name) {
                document.getElementById('user-letter').innerText = user.first_name.charAt(0).toUpperCase();
            }
        }

        // Действие при клике на кнопку
        document.getElementById('main-action').addEventListener('click', () => {
            tg.showAlert('Вы нажали главную кнопку приложения!');
        });
